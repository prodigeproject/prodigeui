PROMPT — REBUILD THE "PICTURA" WEBSITE (PIXEL PERFECT)

Build a single-page website called "Pictura". It is a full-screen dark-mode landing page (background #0E0E0E) where a photograph mosaic in the shape of a heart occupies the center of the screen. As the user scrolls (mouse wheel or touch), the heart's large cards disperse to fixed final positions around the viewport while a small pixel-heart in the center expands and reveals more mini-photos. When any dispersed card is clicked, it expands into a centered modal with metadata. There is no vertical page scroll — scroll events drive the animation state (progress 0..1), not the document.

INTERACTIONS / STATE
- window.addEventListener("wheel", ..., {passive:false}) and touchstart/touchmove accumulate deltaY into `accum` (0..600) and drive progress = accum/600. e.preventDefault() to consume the scroll. Disabled while a modal is open.
- progress drives: heart cells' linear interpolation from initial layout to per-photo TARGETS map (see the TARGETS object in the source; contains left/right/top/bottom or a raw translate() for each dispersed filename), non-targeted cells fly off in their scatter vector; the small heart scale, background color, and expand-image opacity; the bottom text opacity.
- Parallax: only active after progress >= 1. Per-cell multiplier 5..35px inversely proportional to card area, with deterministic sin-based jitter of ±25%. requestAnimationFrame loop, easing current toward target at 0.08 per frame.
- Clicking a big card only opens the modal when progress >= 0.95.

type SmallCell = { r: number; c: number; x: number; y: number; img?: string; cheb: number; snake: number };

function buildSmallHeart(): SmallCell[] {
  const cells: SmallCell[] = [];
  SMALL_HEART_ROWS.forEach((count, r) => {
    let rowW = count * SMALL_CELL + (count - 1) * SMALL_GAP;
    if (r === 0) rowW += SMALL_ROW1_NOTCH - SMALL_GAP;
    const startX = (SMALL_HEART_W - rowW) / 2;
    const y = r * (SMALL_CELL + SMALL_GAP);
    let x = startX;
    for (let c = 0; c < count; c++) {
      const cheb = Math.max(Math.abs(r - 4), Math.abs(c - 6));
      cells.push({ r, c, x, y, img: SMALL_IMAGES[`${r}-${c}`], cheb, snake: 0 });
      const gap = r === 0 && c === 2 ? SMALL_ROW1_NOTCH : SMALL_GAP;
      x += SMALL_CELL + gap;
    }
  });
  // Serpentine snake ordering for image reveal
  let snakeIdx = 0;
  SMALL_HEART_ROWS.forEach((count, r) => {
    const order = r % 2 === 0
      ? Array.from({ length: count }, (_, i) => i)
      : Array.from({ length: count }, (_, i) => count - 1 - i);
    for (const c of order) {
      const cell = cells.find((k) => k.r === r && k.c === c)!;
      if (cell.img) cell.snake = snakeIdx++;
    }
  });
  return cells;
}

function buildCells(): { cells: Cell[]; heartWidth: number; heartHeight: number } {
  const rowWidths = ROWS.map((row) => {
    const w = row.width ?? row.size;
    let total = row.count * w;
    for (let i = 0; i < row.count - 1; i++) {
      total += row.gaps?.[i] ?? H_GAP;
    }
    return total;
  });
  const heartWidth = Math.max(...rowWidths);

  let y = 0;
  let id = 0;
  const cells: Cell[] = [];
  for (let r = 0; r < ROWS.length; r++) {
    const row = ROWS[r];
    const w = row.width ?? row.size;
    const h = row.size;
    const rowW = rowWidths[r];
    let x = (heartWidth - rowW) / 2;
    for (let c = 0; c < row.count; c++) {
      const seed = (r * 131 + c * 17 + 7) * 9301;
      const rand = (n: number) => {
        const v = Math.sin(seed + n) * 43758.5453;
        return v - Math.floor(v);
      };
      const angle = rand(1) * Math.PI * 2;
      const dist = 420 + rand(2) * 400;
      const img = PERSON_IMAGES[id % PERSON_IMAGES.length];
      cells.push({
        id,
        row: r,
        col: c,
        rowCount: row.count,
        x,
        y,
        w,
        h,
        scatterX: Math.cos(angle) * dist,
        scatterY: Math.sin(angle) * dist * 0.7,
        scatterR: (rand(3) - 0.5) * 60,
        img,
      });
      id++;
      x += w + (row.gaps?.[c] ?? H_GAP);
    }
    y += h + V_GAP;
  }
  const heartHeight = y - V_GAP;
  return { cells, heartWidth, heartHeight };
}

// Letter-by-letter animated text
function AnimatedLetters({
  text,
  baseDelay,
  step = 0.05,
  duration = 0.5,
  animationName = "letterIn",
  className,
  style,
  italicIndex,
}: {
  text: string;
  baseDelay: number;
  step?: number;
  duration?: number;
  animationName?: string;
  className?: string;
  style?: React.CSSProperties;
  italicIndex?: number;
}) {
  return (
    <span className={className} style={style}>
      {Array.from(text).map((ch, i) => (
        <span
          key={i}
          style={{
            display: "inline-block",
            opacity: 0,
            fontStyle: italicIndex === i ? "italic" : undefined,
            animation: `${animationName} ${duration}s cubic-bezier(0.22,1,0.36,1) forwards`,
            animationDelay: `${baseDelay + i * step}s`,
            whiteSpace: "pre",
          }}
        >
          {ch === " " ? "\u00A0" : ch}
        </span>
      ))}
    </span>
  );
}

// Animated Get Started button with 2-stage animation
function AnimatedGetStarted({

  variant: "outline" | "filled";

// Crossfades between image src changes with a fade-out + fade-in transition.
function CrossfadeImg({ src, alt, style }: { src: string; alt: string; style?: React.CSSProperties }) {
  const [current, setCurrent] = useState(src);
  const [previous, setPrevious] = useState<string | null>(null);
  const [showCurrent, setShowCurrent] = useState(true);

  useEffect(() => {
    if (src === current) return;
    setPrevious(current);
    setCurrent(src);
    setShowCurrent(false);
    const t = setTimeout(() => setShowCurrent(true), 30);
    return () => clearTimeout(t);
  }, [src, current]);

  useEffect(() => {
    if (!showCurrent || !previous) return;
    const t = setTimeout(() => setPrevious(null), 650);
    return () => clearTimeout(t);
  }, [showCurrent, previous]);

  return (
    <>
      {previous && (
        <img
          src={previous}
          alt={alt}
          draggable={false}
          style={{
            ...style,
            opacity: showCurrent ? 0 : 1,
            transition: "opacity 0.6s ease",
            zIndex: 1,
          }}
        />
      )}
      <img
        src={current}
        alt={alt}
        draggable={false}
        style={{
          ...style,
          opacity: showCurrent ? 1 : 0,
          transition: "opacity 0.6s ease",
          zIndex: 2,
        }}
      />
    </>
  );
}

function SmallHeart({ progress, cycleIndex }: { progress: number; cycleIndex: number }) {

  // Scale from 1 -> (11.267/5.067) as scroll progresses

  const scale = 1 + (targetScale - 1) * p;
  // New images fade in during expansion (0.15 -> 0.85 progress window)
  const expandOpacity = Math.min(1, Math.max(0, (p - 0.15) / 0.55));

  const bgLerp = Math.round(255 - (255 - 217) * p);
  const cubeBg = `rgb(${bgLerp}, ${bgLerp}, ${bgLerp})`;

  // Scroll-triggered image cycling: each cell's index is offset by its snake
  // position so cells swap sequentially one after another, looping forever.
  const step = Math.floor(cycleIndex);

  return (

        top: SMALL_HEART_TOP,

        height: SMALL_HEART_H,

function parseTranslate(str: string): { x: number; y: number } {
  const m = str.match(/translate\(([-\d.]+)px,\s*([-\d.]+)px\)/);
  if (!m) return { x: 0, y: 0 };
  return { x: parseFloat(m[1]), y: parseFloat(m[2]) };
}

function parseVal(v: Val, base: number): number {
  if (typeof v === "number") return v;
  if (typeof v === "string" && v.endsWith("%")) return (parseFloat(v) / 100) * base;
  return parseFloat(String(v));
}
